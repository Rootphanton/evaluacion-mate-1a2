/**
 * EVALUACIÓN DE INTERCAMBIO PEDAGÓGICO 2026 (sobre 45 puntos)

var NOMBRE_HOJA = "Resultados";
var CLAVE_DOCENTE = "ayacucho2026irra"; // clave para ver la tabla de puntos (cámbiala si quieres)
var ENCABEZADOS = ["FECHA Y HORA", "NOMBRES Y APELLIDOS", "CURSO", "PARALELO", "CURSO COMPLETO",
                   "ÁREA", "PUNTAJE (SOBRE 45)", "ACIERTOS (DE 9)", "PREGUNTAS FALLADAS", "DETALLE DE RESPUESTAS"];

function hoja_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(NOMBRE_HOJA);
  if (!sh) {
    sh = ss.insertSheet(NOMBRE_HOJA);
    sh.appendRow(ENCABEZADOS);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, ENCABEZADOS.length).setFontWeight("bold").setBackground("#0f4d2e").setFontColor("#ffffff");
    sh.getRange("A:A").setNumberFormat("@");
  }
  return sh;
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// Recibe la nota desde el examen (sin duplicados si el estudiante reenvía)
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var d = JSON.parse(e.postData.contents);
    var sh = hoja_();
    var fecha = d.fechaHora || new Date().toLocaleString("es-BO");
    var nombre = d.nombre || "";
    var v = sh.getDataRange().getValues();
    for (var i = 1; i < v.length; i++) {
      if (String(v[i][0]) === String(fecha) && String(v[i][1]) === nombre) return json_({ status: "duplicado" });
    }
    sh.appendRow([fecha, nombre, d.curso || "", d.paralelo || "", d.cursoCompleto || "",
                  d.area || "Intercambio Pedagógico", Number(d.puntaje) || 0, d.aciertosTexto || "", d.preguntasFalladas || "", d.detalleRespuestas || ""]);
    return json_({ status: "ok" });
  } finally {
    lock.releaseLock();
  }
}

function datos_() {
  var v = hoja_().getDataRange().getValues();
  v.shift();
  return v.filter(function (r) { return r[1]; }).map(function (r) {
    return { fecha: String(r[0]), nombre: String(r[1]), curso: String(r[4]),
             puntaje: Number(r[6]) || 0, aciertos: String(r[7]) };
  });
}

// Tabla de puntos: abre la URL de la aplicación web
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.clave !== CLAVE_DOCENTE) return pagina_(formulario_());
  if (p.formato === "json") return json_(datos_());
  return pagina_(tabla_(datos_()));
}
